run("Tree Rings (48K)");
run("Size...", "width=898 height=81 constrain interpolation=Bilinear");
rename("in");
n = 40;
for ( i=1; i<n; i++) { 
	selectWindow("in");
	a = 0.5 + (i-n/2)*(i-n/2)*0.5;
	run("OrientationJ Analysis", "tensor=" + a + " hue=Orientation sat=Coherency bri=Original-Image ");
}
run("Images to Stack", "title=Color-survey");
selectWindow("in");
close();
